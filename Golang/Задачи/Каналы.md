
## Задача 3
Что произойдет при текущей реализации функции **spawnMessages**?  почему ?
Каким образом можно исправить работоспособность функции? 
```go
package main

import (
    "fmt"
)

func spawnMessages(n int) chan string {
    resChan := make(chan string, 1)
  
    for i := 0; i < n; i++ {
        resChan <- fmt.Sprintf("msg: %d", i)
    }

    return resChan
}

  

func main() {

    n := 10

    for msg := range spawnMessages(n) {
        fmt.Println("Received, ", msg)
    }
}
```

### исправленная редакция  spawnMessages()
```go 

func spawnMessages(n int) chan string {
    resChan := make(chan string, 1)

    wg := sync.WaitGroup{}
    wg.Add(1)
    go func() {
        defer wg.Done()
        for i := 0; i < n; i++ {
            resChan <- fmt.Sprintf("msg: %d", i)
        }
    }()
  
    go func() {
        wg.Wait()
        close(resChan)
    }()

    return resChan
}
```

## Задача 5
Реализовать функцию- обертку **predictableFunc()** для **unpredictableFunc()**  которая бы ИЛИ возвращала результат от **unpredictableFunc()** ИЛИ прекрашала выполнение не дожидаясь  **unpredictableFunc()**, вернув zero value + ошибку.

```go
package main

import (
    "fmt"
    "time"
    // "errors"
    "context"
    "math/rand"
)
  
// может работать вплоть до 40 сек
func unpredictableFunc() int {
    result := rand.Intn(10)
    time.Sleep(time.Second * time.Duration(result))
    //fmt.Printf("unpredictableFunc() returns: %d\n", result)
    return result
}



func predictableFunc(ctx context.Context) (int, error) {
    okChan := make(chan struct{})
    var res int
  
    // если переданный параметром контекст не имее дедлайна - создаем дефолтный, который закроется через 5 сек
    var cancel context.CancelFunc
    if _, ok := ctx.Deadline(); !ok {
        ctx, cancel = context.WithTimeout(ctx, time.Second*5)
        defer cancel()
    }

    go func(res *int) {
        *res = unpredictableFunc()
        close(okChan)
    }(&res)

  

    select {
    case _ = <-okChan:
        return res, nil
    case _ = <-ctx.Done():
        return 0, ctx.Err()
    }
}

  
func main() {
    res, err := predictableFunc(context.Background())
    fmt.Println("res: ", res)
    fmt.Println("err", err)
}
```
