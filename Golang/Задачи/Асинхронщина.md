

## Задача 1
- Добавить асинронного поведения в функцию `process()` чтобы ускорить ее работу
- в случае полчения первой же ошибки - прекратить дальнейшее исполнение
process() и вернуть первую же ошибку.
```go
package main

import (
    "fmt"
    "context"
    // "sync"
    "time"
)

type User struct {
    Name string
}

// имитация вызова в сеть с задержкой
func fetch(ctx context.Context, u User)( string, error ){
    time.Sleep(time.Millisecond * 10)
    return u.Name, nil
}

func process(ctx context.Context, users []User)(map[string]int, error) {
    names := make(map[string]int, 0)
  
    for _, u := range users {
        name, err := fetch(ctx, u)
        if err != nil{}

        names[name] = names[name] + 1
    }

    return names, nil
}

  

func main() {

    names := []User{    
        {"Anna"},
        {"Bob"},
        {"Cindy"},
        {"Bob"},
    }

    ctx := context.Background()
    start := time.Now()
  
    res, err := process(ctx, names)
    if err != nil {
        fmt.Println("an error occured: ", err.Error())
    }

    fmt.Println("time:", time.Since(start))
    fmt.Println(res)
}
```

### Решение 1
```go
func process(ctx context.Context, users []User) (map[string]int, error) {

    names := make(map[string]int, 0)

    ctx, cancel := context.WithCancel(ctx)
    defer cancel()

    var commonError error

    mu := sync.Mutex{}
    wg := sync.WaitGroup{}

    for _, u := range users {
        wg.Add(1)
        go func() {
            defer wg.Done()
            name, err := fetch(ctx, u)
            if err != nil {
                sync.OnceFunc(func() {
                    cancel()
                    commonError = err
                })()
            }

            mu.Lock()
            names[name] = names[name] + 1
            mu.Unlock()
        }()
    }

    wg.Wait()

    if commonError != nil {
        return nil, commonError
    }
    return names, nil
}
```


### решение 2 (с применением errgroup)
```go
func process(ctx context.Context, users []User) (map[string]int, error) {

    names := make(map[string]int, 0)

    ctx, cancel := context.WithCancel(ctx)
    defer cancel()
  
    mu := sync.Mutex{}

    errgroup, ectx := errgroup.WithContext(ctx)
    errgroup.SetLimit(10)
    for _, u := range users {

        errgroup.Go(func() error {

            name, err := fetch(ectx, u)
            if err != nil {
                return err
            }
            mu.Lock()
            names[name] = names[name] + 1
            mu.Unlock()

            return nil
        })
    }

    if err := errgroup.Wait(); err != nil {
        return nil, err
    }

    return names, nil

}
```
