```go
package main

  

import (
    "context"
    "fmt"
    "sync"
    "time"
)

type Cache struct {
    data map[string]string
}

type CacheI interface {
    Get(ctx context.Context, key string) (string, error)
    Set(ctx context.Context, key string, value string) error
    Delete(ctx context.Context, key string) error
    Stop()
}

  

var NotFound = fmt.Errorf("key not found")

// единица хранения в кэше
type elem struct {
    value    string    // сами данные
    exp_date time.Time // дата/время истечения срока годности
}

  
  

// кэш с TTL

type cache struct {
    data map[string]elem
    mu   *sync.Mutex
    ttl  time.Duration
    done chan struct{}
}

  
  
  

// конструктор кэша
func New(ttl time.Duration) *cache {
    result := &cache{
        data: make(map[string]elem),
        mu:   &sync.Mutex{},
        ttl:  ttl,
        done: make(chan struct{}),
    }

    result.clearByTTL()
    return result
}

  

func (c *cache) Get(_ context.Context, key string) (string, error) {
    c.mu.Lock()
    // defer c.mu.Unlock()
  
    elem, ok := c.data[key]

    c.mu.Unlock()

    if !ok {
        return "", NotFound
    }
  
    if elem.exp_date.Before(time.Now()) {
        c.delete(key)
        return "", NotFound
    }

    return elem.value, nil
}

  

func (c *cache) Set(_ context.Context, key string, value string) error {
    c.mu.Lock()
    // defer c.mu.Unlock()
    elem := elem{
        value:    value,
        exp_date: time.Now().Add(c.ttl),
    }

    c.data[key] = elem
    c.mu.Unlock()
    return nil
}

  

func (c *cache) Delete(_ context.Context, key string) error {

    c.delete(key)

    return nil
}

  

func (c *cache) delete(key string) {

    c.mu.Lock()
    // defer c.mu.Unlock()
    delete(c.data, key)
    c.mu.Unlock()

}

  
  

func(c *cache) clear() {
    c.mu.Lock()
    defer c.mu.Unlock()
    for key, elem := range c.data {
        if elem.exp_date.Before(time.Now()) {

            // используем стандартный delete, а не метод, для того чтобы мьютекс лочисля лишь единожды на все итерации удаления
            delete(c.data, key)
        }
    }
    c.mu.Unlock()
}


func (c *cache) clearByTTL() {
    ticker := time.NewTicker(time.Second * 10) // запуск - каждые 10 секунд
    go func(){
        for {
        select {
        case <- ticker.C:
            c.clear()
        case <- c.done:
            return
        }
    }
    }()
}

  

func (c *cache) Stop()  {
    close(c.done)
}
```
